            if (isPremiumAdFree()) return;
            document.querySelectorAll('.section-divider-ad[data-ad-network="hilltop"]').forEach((wrapper) => {
                const stage = wrapper.querySelector(".combo-ad-stage");
                if (!stage) return;

                const hasVisualAd = () => {
                    const els = stage.querySelectorAll("iframe, ins, img, a, object, div");
                    for (const el of els) {
                        const rect = el.getBoundingClientRect();
                        if (rect.width >= 50 && rect.height >= 40) return true;
                    }
                    return false;
                };

                const obs = new MutationObserver(() => {
                    if (hasVisualAd()) wrapper.style.display = "flex";
                });
                obs.observe(stage, { childList: true, subtree: true, attributes: true });

                setTimeout(() => {
                    if (!hasVisualAd()) wrapper.style.display = "none";
                }, 7500);
            });

        window.toggleMyList = async function (btnElement, titleSafe, imgSafe, linkSafe, rawId) {
            if (!rawId) return;

            let profileMyList = [];
            let watchlistObj = {};
            try {
                profileMyList = JSON.parse(localStorage.getItem("dramakan_mylist")) || [];
                watchlistObj = JSON.parse(localStorage.getItem("dramakan_watchlist")) || {};
            } catch {}

            const title = decodeURIComponent(titleSafe);
            const img = decodeURIComponent(imgSafe);
            const link = decodeURIComponent(linkSafe);

            const inListIdx = profileMyList.findIndex((item) => String(item.id) === String(rawId));
            let added = false;

            if (inListIdx > -1) {
                profileMyList.splice(inListIdx, 1);
                delete watchlistObj[rawId];
                btnElement.classList.remove("active");
                btnElement.innerHTML = btnElement.classList.contains("btn-mylist")
                    ? '<i class="fas fa-plus"></i> <span>My List</span>'
                    : '<i class="fas fa-plus"></i>';
                showToast(`Removed "${title}" from My List`);
            } else {
                added = true;
                const itemData = { id: String(rawId), title, img, link, timestamp: Date.now() };
                profileMyList.push(itemData);
                watchlistObj[rawId] = itemData;
                btnElement.classList.add("active");
                btnElement.innerHTML = btnElement.classList.contains("btn-mylist")
                    ? '<i class="fas fa-check"></i> <span>In List</span>'
                    : '<i class="fas fa-check"></i>';
                showToast(`Added "${title}" to My List`);
            }

            localStorage.setItem("dramakan_mylist", JSON.stringify(profileMyList));
            localStorage.setItem("dramakan_watchlist", JSON.stringify(watchlistObj));

            if (firebaseInstance && firebaseInstance.auth.currentUser) {
                try {
                    const user = firebaseInstance.auth.currentUser;
                    const { doc, getDoc, updateDoc } = firebaseInstance.firestoreModule;
                    const userRef = doc(firebaseInstance.db, "users", user.uid);
                    const snap = await getDoc(userRef);
                    if (snap.exists()) {
                        const data = snap.data();
                        if (data.profiles && data.profiles.length > 0) {
                            const activeId = localStorage.getItem("dramakan_active_profile_id");
                            let pIdx = data.profiles.findIndex((p) => p.id === activeId);
                            if (pIdx === -1) pIdx = 0;
                            data.profiles[pIdx].myList = profileMyList;
                            await updateDoc(userRef, { profiles: data.profiles });
                        } else {
                            await updateDoc(userRef, { myList: profileMyList });
                        }
                    }
                } catch {}
            }
            return added;
        };

        async function renderContinueWatching(cloudHistoryArr = null) {

                        .filter((item) => item && item.link && item.title && !item.link.toLowerCase().includes("index.html"))

        function buildNextCategoryDefinition() {
            const list = getFilteredCatalog();
            if (!list.length) return null;

            if (nextCatalogIndex < list.length) {
                const base = list[nextCatalogIndex++];
                return {
                    ...base,
                    uniqueId: `${base.id}-w${currentWave}-${nextCatalogIndex}`,
                    fetchUrl: `${TMDB_API}${base.endpoint}${base.endpoint.includes("?") ? "&" : "?"}api_key=${tmdbKey}&page=1`
                };
            }

            // Seamless Endless Loop: move to next Wave (TMDB page=2, 3, 4...) so the user never hits a dead end
            currentWave++;
            nextCatalogIndex = 0;
            const base = list[nextCatalogIndex++];

        const shelfDataObserver = new IntersectionObserver(
            (entries, observer) => {
                entries.forEach(async (entry) => {
                    if (!entry.isIntersecting) return;
                    const gridId = entry.target.id;
                    const cfg = shelfConfigRegistry.get(gridId);
                    if (!cfg) return;
                    observer.unobserve(entry.target);

                    try {
                        const res = await fetch(cfg.fetchUrl);
                        if (!res.ok) throw new Error("Fetch failed");
                        const data = await res.json();
                        const results = (data.results || []).filter((item) => item && (item.poster_path || item.backdrop_path)).slice(0, 16);

                        if (!results.length) {
                            entry.target.closest(".latest-episodes")?.remove();
                            return;
                        }

                        const mapped = results.map((i) => {
                            const year = (i.release_date || i.first_air_date || "").substring(0, 4);
                            const rating = i.vote_average && i.vote_average > 0 ? i.vote_average.toFixed(1) : "";
                            return {
                                id: String(i.id),
                                title: i.name || i.title || "Untitled",
                                img: i.poster_path
                                    ? `${TMDB_IMG}/w342${i.poster_path}`
                                    : `${TMDB_IMG}/w500${i.backdrop_path}`,
                                link: `watch.html?id=${i.id}`,
                                type: year ? `${year} • ${cfg.tag}` : cfg.tag,
                                rating,
                                media_type: cfg.type

                        });

                });

        function spawnCategoryBatch(batchSize = 4) {
            if (isBatchSpawning || !infiniteRoot) return;
            isBatchSpawning = true;

            const frag = document.createDocumentFragment();
            const newlyCreatedGrids = [];

            for (let i = 0; i < batchSize; i++) {
                const cfg = buildNextCategoryDefinition();
                if (!cfg) break;

                totalMountedShelves++;
                cfg.allowInlineAd = totalMountedShelves <= 18 && totalMountedShelves % 3 === 0;

                const { section, grid } = createCategoryShelfElement(cfg);
                shelfConfigRegistry.set(cfg.uniqueId, cfg);
                frag.appendChild(section);
                newlyCreatedGrids.push(grid);

                // Insert a hybrid HilltopAds / Adsterra combo divider every 3 shelves
                if (totalMountedShelves % 3 === 0) {
                    const adEl = createComboAdDivider();
                    if (adEl) frag.appendChild(adEl);
                }
            }

            infiniteRoot.appendChild(frag);
            newlyCreatedGrids.forEach((grid) => shelfDataObserver.observe(grid));

            updateCounterChip();
            if (endlessStatusText) {
                endlessStatusText.textContent = `Endless Discovery Active • ${totalMountedShelves} categories loaded (keep scrolling!)`;
            }

            setTimeout(() => {
                isBatchSpawning = false;
            }, 120);
        }

        const bindDynamicSearch = (inputId, resultsId) => {
            const searchInput = document.getElementById(inputId);
            const searchResults = document.getElementById(resultsId);
            if (!searchInput || !searchResults) return;

            let debounceTimer = null;
            let activeSearchToken = 0;

            searchInput.addEventListener("input", () => {
                clearTimeout(debounceTimer);
                const queryStr = searchInput.value.trim();
                if (!queryStr) {
                    searchResults.style.display = "none";
                    return;
                }

                debounceTimer = setTimeout(async () => {
                    const token = ++activeSearchToken;
                    try {
                        const res = await fetch(`${TMDB_API}/search/multi?api_key=${tmdbKey}&query=${encodeURIComponent(queryStr)}`);
                        if (!res.ok || token !== activeSearchToken) return;
                        const data = await res.json();
                        const resultsMapped = (data.results || [])
                            .filter((i) => i.media_type === "tv" || i.media_type === "movie")
                            .slice(0, 8);

                        searchResults.innerHTML = "";
                        if (!resultsMapped.length) {
                            const empty = document.createElement("div");
                            empty.style.cssText = "padding: 14px; text-align: center; color: var(--text-muted); font-size: 0.84rem;";
                            empty.textContent = "No results found";
                            searchResults.appendChild(empty);
                        } else {
                            const frag = document.createDocumentFragment();
                            resultsMapped.forEach((item) => {
                                const a = document.createElement("a");
                                a.href = `watch.html?id=${encodeURIComponent(item.id)}`;
                                a.className = "search-result-item";

                                const img = document.createElement("img");
                                img.src = item.poster_path ? `${TMDB_IMG}/w92${item.poster_path}` : "default-poster.jpg";
                                img.alt = item.name || item.title || "Poster";
                                img.loading = "lazy";
                                img.decoding = "async";

                                const info = document.createElement("div");
                                info.className = "search-result-info";
                                const titleDiv = document.createElement("div");
                                titleDiv.className = "search-result-title";
                                titleDiv.textContent = item.name || item.title || "Untitled";
                                const small = document.createElement("small");
                                const yr = (item.release_date || item.first_air_date || "").substring(0, 4);
                                small.textContent = yr ? `${item.media_type.toUpperCase()} • ${yr}` : item.media_type.toUpperCase();
                                info.append(titleDiv, small);

                                a.append(img, info);
                                frag.appendChild(a);
                            });
                            searchResults.appendChild(frag);
                        }
                        searchResults.style.display = "block";
                    } catch {}
                }, 280);
            });

            document.addEventListener("click", (e) => {
                if (!searchInput.contains(e.target) && !searchResults.contains(e.target)) {
                    searchResults.style.display = "none";
                }
            });
        };

        bindDynamicSearch("searchInput", "searchResults");
        bindDynamicSearch("mobileSearchInput", "mobileSearchResults");

        function createProfileSwitcher(profiles) {
            if (document.getElementById("home-profile-switcher-overlay")) return;
            const overlayContainer = document.createElement("div");
            overlayContainer.id = "home-profile-switcher-overlay";

            const title = document.createElement("h2");
            title.className = "switcher-title";
            title.textContent = "Who's watching?";

            const listDiv = document.createElement("div");
            listDiv.className = "profiles-list";

            profiles.forEach((prof) => {
                const card = document.createElement("div");
                card.className = "profile-select-card";
                card.dataset.id = prof.id;

                const isEmoji = prof.avatar && prof.avatar.length <= 10 && !prof.avatar.includes("http") && !prof.avatar.includes("data:image");
                if (isEmoji) {
                    const av = document.createElement("div");
                    av.className = "switcher-avatar-img";
                    av.textContent = prof.avatar;
                    card.appendChild(av);
                } else {
                    const av = document.createElement("img");
                    av.className = "switcher-avatar-img";
                    av.src = prof.avatar;
                    av.alt = prof.name;
                    card.appendChild(av);
                }

                const nameSpan = document.createElement("span");
                nameSpan.className = "switcher-name";
                nameSpan.textContent = prof.name;
                card.appendChild(nameSpan);

                card.addEventListener("click", () => {
                    localStorage.setItem("dramakan_profile_prompt_date", new Date().toDateString());
                    localStorage.setItem("dramakan_active_profile_id", prof.id);
                    overlayContainer.classList.add("hidden");
                    setTimeout(() => overlayContainer.remove(), 450);
                    window.dispatchEvent(new CustomEvent("profileSelected"));
                });

                listDiv.appendChild(card);
            });

            overlayContainer.append(title, listDiv);
            document.body.appendChild(overlayContainer);
        }

        function updateHeaderAvatar(profiles) {
            if (!profiles.length) return;
            const activeId = localStorage.getItem("dramakan_active_profile_id");
            const activeProf = profiles.find((p) => p.id === activeId) || profiles[0];
            const avatarUrl = activeProf.avatar;
            const isEmoji = avatarUrl && avatarUrl.length <= 10 && !avatarUrl.includes("http") && !avatarUrl.includes("data:image");

            const authBtn = document.getElementById("topAuthBtn");
            const bottomAuthBtn = document.getElementById("bottomAuthBtn");

            if (authBtn) {
                authBtn.href = "profile.html";
                authBtn.style.cssText = "padding: 0 !important; background: transparent !important; border: none !important; box-shadow: none !important;";
                authBtn.innerHTML = isEmoji
                    ? `<div style="width: 40px; height: 40px; border-radius: 50%; background: rgba(255,255,255,0.1); border: 2px solid var(--primary-bright); display: flex; align-items: center; justify-content: center; font-size: 1.3rem;">${avatarUrl}</div>`
                    : `<img src="${avatarUrl}" alt="Profile" style="width: 40px; height: 40px; border-radius: 50%; object-fit: cover; border: 2px solid var(--primary-bright);">`;
            }

            if (bottomAuthBtn) {
                const navIcon = bottomAuthBtn.querySelector(".nav-icon");
                if (navIcon) {
                    navIcon.outerHTML = isEmoji
                        ? `<div class="nav-icon" style="width: 26px; height: 26px; border-radius: 50%; display: flex; align-items: center; justify-content: center; font-size: 1rem; border: 1px solid var(--primary-bright); color: #fff;">${avatarUrl}</div>`
                        : `<img src="${avatarUrl}" alt="Profile" class="nav-icon" style="width: 26px; height: 26px; border-radius: 50%; object-fit: cover; border: 1px solid var(--primary-bright);">`;
                }
                bottomAuthBtn.href = "profile.html";
            }
        }

        async function initAuthSync() {
            try {
                const { auth, db, firestoreModule, authModule } = await getFirebase();
                const { doc, collection, onSnapshot } = firestoreModule;
                const { onAuthStateChanged } = authModule;

                onAuthStateChanged(auth, async (user) => {
                    if (user) {
                        try {
                            const sub = await getSubscriptionStatus(db, user.uid);
                            applySubscriptionPrivileges(sub);
                        } catch {}

                        try {
                            onSnapshot(doc(db, "users", user.uid), (docSnap) => {
                                let userProfiles = [];
                                let legacyAvatar = `https://api.dicebear.com/7.x/adventurer/svg?seed=${encodeURIComponent(user.displayName || user.email || "User")}`;

                                if (docSnap.exists()) {
                                    const data = docSnap.data();
                                    if (data.avatarUrl) legacyAvatar = data.avatarUrl;
                                    userProfiles =
                                        data.profiles && data.profiles.length > 0
                                            ? data.profiles
                                            : [{ id: "prof_default", name: data.username || "User", avatar: legacyAvatar }];
                                }

                                const todayStr = new Date().toDateString();
                                const lastPromptDate = localStorage.getItem("dramakan_profile_prompt_date");
                                if (lastPromptDate !== todayStr && userProfiles.length > 0) {
                                    createProfileSwitcher(userProfiles);
                                } else {
                                    updateHeaderAvatar(userProfiles);
                                }

                                window.addEventListener("profileSelected", () => updateHeaderAvatar(userProfiles), { once: true });
                            });

                            const historyRef = collection(db, "users", user.uid, "history");
                            if (unsubscribeHistory) unsubscribeHistory();

                            unsubscribeHistory = onSnapshot(historyRef, (snapshot) => {
                                const cloudHistory = {};
                                snapshot.forEach((docItem) => {
                                    const histData = docItem.data();
                                    if (histData && histData.dramaId) {
                                        histData.id = String(histData.dramaId);
                                        cloudHistory[histData.id] = histData;
                                    }
                                });

                                localStorage.setItem("dramakan_history", JSON.stringify(cloudHistory));
                                const historyArr = Object.values(cloudHistory)
                                    .filter((item) => item && item.link && item.title && !item.link.toLowerCase().includes("index.html"))
                                    .sort((a, b) => (b.timestamp || 0) - (a.timestamp || 0))
                                    .slice(0, 12);

                                renderContinueWatching(historyArr);
                            });
                        } catch {}
                    } else {
                        const authBtn = document.getElementById("topAuthBtn");
                        if (authBtn) {
                            authBtn.href = "login.html";
                            authBtn.innerHTML = '<i class="fas fa-user"></i> <span>Login / Sign Up</span>';
                            authBtn.style.cssText = "";
                        }
                        if (unsubscribeHistory) {
                            unsubscribeHistory();
                            unsubscribeHistory = null;
                        }
                        renderContinueWatching();
                    }
                });
            } catch {
                renderContinueWatching();
            }
        }
